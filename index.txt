    <!-- MAIN APP SCRIPT -->
    <script>
        // CONFIGURAÇÃO DO GOOGLE SHEETS (COLOCA AQUI O TEU URL /exec)
        const SHEETS_URL = 'https://script.google.com/macros/s/SEU_SCRIPT_ID_AQUI/exec';

        const fieldsOrder = ['piq_litros', 'piq_ec', 'piq_ph', 'b1_litros', 'b1_ec', 'b1_ph', 'b2_litros', 'b2_ec', 'b2_ph'];
        let activeField = 'piq_litros';
        let currentValve = 1;
        let formValues = {
            piq_litros: '', piq_ec: '', piq_ph: '',
            b1_litros: '', b1_ec: '', b1_ph: '',
            b2_litros: '', b2_ec: '', b2_ph: ''
        };

        let piquetasConfig = {};
        let recordsData = [];

        window.onload = function() {
            loadPiquetasConfig();
            loadRecords();
            renderValveButtons();
            renderPiquetasConfigUI();
            setActiveField('piq_litros');
            
            const hoje = new Date().toISOString().split('T')[0];
            document.getElementById('hist-start').value = hoje;
            document.getElementById('hist-end').value = hoje;
            document.getElementById('today-date-label').innerText = formatDatePt(hoje);

            renderTodayTable();
            sincronizarComSheets();
            setInterval(sincronizarComSheets, 30000); // Tenta sincronizar a cada 30s em segundo plano
        };

        // --- NAVEGAÇÃO ENTRE ABAS ---
        function switchTab(tab) {
            ['registos', 'historico', 'piquetas'].forEach(t => {
                document.getElementById(`tab-${t}`).classList.add('hidden');
                const btn = document.getElementById(`tab-btn-${t}`);
                btn.className = "flex-1 py-3 text-center border-b-4 border-transparent text-emerald-200 hover:text-white flex items-center justify-center gap-2";
            });
            document.getElementById(`tab-${tab}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`tab-btn-${tab}`);
            activeBtn.className = "flex-1 py-3 text-center border-b-4 border-emerald-400 text-white flex items-center justify-center gap-2 font-bold";

            if(tab === 'historico') renderHistoryTable();
        }

        // --- GERIR VÁLVULAS ---
        function renderValveButtons() {
            const container = document.getElementById('valve-buttons-container');
            container.innerHTML = '';
            for(let i=1; i<=10; i++) {
                const isSelected = i === currentValve;
                container.innerHTML += `
                    <button onclick="selectValve(${i})" class="py-3 px-1 rounded-xl font-black text-sm transition-all shadow-sm border ${isSelected ? 'bg-emerald-600 text-white border-emerald-700 shadow-md scale-105' : 'bg-slate-50 text-slate-700 border-slate-200 hover:bg-slate-100'}">
                        V${i}
                    </button>
                `;
            }
        }

        function selectValve(v) {
            currentValve = v;
            renderValveButtons();
            document.getElementById('selected-valve-badge').innerText = `Válvula ${v}`;
            const np = piquetasConfig[`v${v}`] || 10;
            document.getElementById('valve-pins-indicator').innerText = `(${np} piquetas)`;
            updateLiveDrainages();
        }

        // --- CONFIGURAÇÃO DE PIQUETAS ---
        function loadPiquetasConfig() {
            const saved = localStorage.getItem('agro_piquetas_cfg');
            if(saved) {
                try { piquetasConfig = JSON.parse(saved); } catch(e) { piquetasConfig = {}; }
            }
            for(let i=1; i<=10; i++) {
                if(!piquetasConfig[`v${i}`]) piquetasConfig[`v${i}`] = 10;
            }
        }

        function renderPiquetasConfigUI() {
            const container = document.getElementById('piquetas-config-container');
            container.innerHTML = '';
            for(let i=1; i<=10; i++) {
                container.innerHTML += `
                    <div class="flex items-center justify-between bg-slate-50 p-3.5 rounded-xl border border-slate-200">
                        <span class="font-bold text-sm text-slate-700">Válvula ${i}</span>
                        <div class="flex items-center gap-2">
                            <label class="text-xs text-slate-500 font-semibold">Piquetas:</label>
                            <input type="number" id="cfg-v${i}" value="${piquetasConfig[`v${i}`]}" min="1" class="w-20 text-center font-bold text-base bg-white border border-slate-300 rounded-lg py-1">
                        </div>
                    </div>
                `;
            }
        }

        function savePiquetasConfig() {
            for(let i=1; i<=10; i++) {
                const val = parseInt(document.getElementById(`cfg-v${i}`).value) || 10;
                piquetasConfig[`v${i}`] = val;
            }
            localStorage.setItem('agro_piquetas_cfg', JSON.stringify(piquetasConfig));
            selectValve(currentValve);
            showToast('Configuração de piquetas guardada!');
            renderTodayTable();
        }

        // --- INPUTS E TECLADO TÁTIL ---
        function setActiveField(fieldName) {
            activeField = fieldName;
            fieldsOrder.forEach(f => {
                const el = document.getElementById(`input-${f}`);
                if(el) el.classList.remove('field-active');
            });
            const activeEl = document.getElementById(`input-${fieldName}`);
            if(activeEl) activeEl.classList.add('field-active');
        }

        function keypadInput(digit) {
            let val = formValues[activeField] || '';
            if(val.length >= 5) return; // Limite razoável
            val += digit;
            formValues[activeField] = val;
            updateFieldDisplay(activeField);
            updateLiveDrainages();
        }

        function keypadBackspace() {
            let val = formValues[activeField] || '';
            if(val.length > 0) {
                val = val.slice(0, -1);
                formValues[activeField] = val;
                updateFieldDisplay(activeField);
                updateLiveDrainages();
            }
        }

        function keypadClearField() {
            formValues[activeField] = '';
            updateFieldDisplay(activeField);
            updateLiveDrainages();
        }

        function updateFieldDisplay(fieldName) {
            const raw = formValues[fieldName];
            const el = document.getElementById(`input-${fieldName}`);
            if(!el) return;
            if(!raw) {
                el.value = '';
                return;
            }
            // Decimais implícitos: 7 vira 0.7, 70 vira 7.0
            const num = parseInt(raw, 10) || 0;
            el.value = (num / 10).toFixed(1);
        }

        function nextField() {
            const idx = fieldsOrder.indexOf(activeField);
            if(idx < fieldsOrder.length - 1) {
                setActiveField(fieldsOrder[idx + 1]);
            } else {
                setActiveField(fieldsOrder[0]); // Volta ao início
            }
        }

        function resetCurrentForm() {
            fieldsOrder.forEach(f => {
                formValues[f] = '';
                updateFieldDisplay(f);
            });
            setActiveField(fieldsOrder[0]);
            updateLiveDrainages();
        }

        // --- CÁLCULO DE DRENAGEM ---
        function calculateDrainage(baldeLitrosNum, piqLitrosNum, numPiquetas) {
            if(!piqLitrosNum || !numPiquetas || piqLitrosNum <= 0 || numPiquetas <= 0 || baldeLitrosNum < 0) return 0;
            const totalPiq = numPiquetas * piqLitrosNum;
            if(totalPiq === 0) return 0;
            return Math.round((baldeLitrosNum / totalPiq) * 100);
        }

        function updateLiveDrainages() {
            const np = piquetasConfig[`v${currentValve}`] || 10;
            
            const pL = parseFloat(document.getElementById('input-piq_litros').value) || 0;
            const b1L = parseFloat(document.getElementById('input-b1_litros').value) || 0;
            const b2L = parseFloat(document.getElementById('input-b2_litros').value) || 0;

            const dr1 = calculateDrainage(b1L, pL, np);
            const dr2 = calculateDrainage(b2L, pL, np);

            document.getElementById('live-dr1').innerText = `DR1: ${dr1}%`;
            document.getElementById('live-dr2').innerText = `DR2: ${dr2}%`;
        }

        // --- GUARDAR E SINCRONIZAR REGISTOS ---
        function loadRecords() {
            const saved = localStorage.getItem('agro_records');
            if(saved) {
                try { recordsData = JSON.parse(saved); } catch(e) { recordsData = []; }
            }
        }

        function saveRecord() {
            const np = piquetasConfig[`v${currentValve}`] || 10;
            
            const pL = parseFloat(document.getElementById('input-piq_litros').value) || 0;
            const b1L = parseFloat(document.getElementById('input-b1_litros').value) || 0;
            const b2L = parseFloat(document.getElementById('input-b2_litros').value) || 0;

            const dr1 = calculateDrainage(b1L, pL, np);
            const dr2 = calculateDrainage(b2L, pL, np);

            const agora = new Date();
            const dataIso = agora.toISOString().split('T')[0];
            const horaStr = agora.toLocaleTimeString('pt-PT', {hour: '2-digit', minute:'2-digit', second:'2-digit'});

            const record = {
                id: 'REG-' + Date.now() + '-' + Math.floor(Math.random()*1000),
                dataIso: dataIso,
                dataHora: `${dataIso} ${horaStr}`,
                valvula: currentValve,
                piquetas: np,
                piq_litros: pL,
                piq_ec: parseFloat(document.getElementById('input-piq_ec').value) || 0,
                piq_ph: parseFloat(document.getElementById('input-piq_ph').value) || 0,
                b1_litros: b1L,
                b1_ec: parseFloat(document.getElementById('input-b1_ec').value) || 0,
                b1_ph: parseFloat(document.getElementById('input-b1_ph').value) || 0,
                dr1: dr1,
                b2_litros: b2L,
                b2_ec: parseFloat(document.getElementById('input-b2_ec').value) || 0,
                b2_ph: parseFloat(document.getElementById('input-b2_ph').value) || 0,
                dr2: dr2,
                sync: false
            };

            recordsData.push(record);
            localStorage.setItem('agro_records', JSON.stringify(recordsData));

            resetCurrentForm();
            renderTodayTable();
            showToast('Registo guardado com sucesso!');
            sincronizarComSheets();
        }

        function renderTodayTable() {
            const hoje = new Date().toISOString().split('T')[0];
            const tbody = document.getElementById('today-table-body');
            const doDia = recordsData.filter(r => r.dataIso === hoje);

            document.getElementById('today-count-badge').innerText = `${doDia.length} registos`;

            if(doDia.length === 0) {
                tbody.innerHTML = `<tr><td colspan="14" class="text-center py-4 text-slate-400">Nenhum registo efetuado hoje.</td></tr>`;
                return;
            }

            tbody.innerHTML = '';
            doDia.reverse().forEach((r, idx) => {
                const hora = r.dataHora.split(' ')[1] || '';
                tbody.innerHTML += `
                    <tr>
                        <td class="font-bold">V${r.valvula}</td>
                        <td>${hora}</td>
                        <td>${r.piq_litros.toFixed(1)}</td>
                        <td>${r.piq_ec.toFixed(1)}</td>
                        <td>${r.piq_ph.toFixed(1)}</td>
                        <td>${r.b1_litros.toFixed(1)}</td>
                        <td>${r.b1_ec.toFixed(1)}</td>
                        <td>${r.b1_ph.toFixed(1)}</td>
                        <td class="bg-blue-50 font-black text-blue-900">${r.dr1}%</td>
                        <td>${r.b2_litros.toFixed(1)}</td>
                        <td>${r.b2_ec.toFixed(1)}</td>
                        <td>${r.b2_ph.toFixed(1)}</td>
                        <td class="bg-purple-50 font-black text-purple-900">${r.dr2}%</td>
                        <td>
                            <button onclick="deleteRecord('${r.id}')" class="text-red-500 hover:text-red-700 p-1"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    </tr>
                `;
            });
        }

        function deleteRecord(id) {
            if(!confirm('Pretende apagar este registo?')) return;
            recordsData = recordsData.filter(r => r.id !== id);
            localStorage.setItem('agro_records', JSON.stringify(recordsData));
            renderTodayTable();
            showToast('Registo removido.');
        }

        function renderHistoryTable() {
            const valvFiltro = document.getElementById('hist-valve').value;
            const startFiltro = document.getElementById('hist-start').value;
            const endFiltro = document.getElementById('hist-end').value;

            const tbody = document.getElementById('history-table-body');
            
            let filtrados = recordsData.filter(r => {
                if(valvFiltro !== 'ALL' && String(r.valvula) !== valvFiltro) return false;
                if(startFiltro && r.dataIso < startFiltro) return false;
                if(endFiltro && r.dataIso > endFiltro) return false;
                return true;
            });

            if(filtrados.length === 0) {
                tbody.innerHTML = `<tr><td colspan="14" class="text-center py-4 text-slate-400">Sem registos encontrados para o filtro aplicado.</td></tr>`;
                return;
            }

            tbody.innerHTML = '';
            filtrados.reverse().forEach(r => {
                const dataPt = formatDatePt(r.dataIso);
                const hora = r.dataHora.split(' ')[1] || '';
                tbody.innerHTML += `
                    <tr>
                        <td class="font-semibold">${dataPt}</td>
                        <td>${hora}</td>
                        <td class="font-bold">V${r.valvula}</td>
                        <td>${r.piq_litros.toFixed(1)}</td>
                        <td>${r.piq_ec.toFixed(1)}</td>
                        <td>${r.piq_ph.toFixed(1)}</td>
                        <td>${r.b1_litros.toFixed(1)}</td>
                        <td>${r.b1_ec.toFixed(1)}</td>
                        <td>${r.b1_ph.toFixed(1)}</td>
                        <td class="bg-blue-50 font-black text-blue-900">${r.dr1}%</td>
                        <td>${r.b2_litros.toFixed(1)}</td>
                        <td>${r.b2_ec.toFixed(1)}</td>
                        <td>${r.b2_ph.toFixed(1)}</td>
                        <td class="bg-purple-50 font-black text-purple-900">${r.dr2}%</td>
                    </tr>
                `;
            });
        }

        // --- SINCRONIZAÇÃO COM GOOGLE SHEETS ---
        async function sincronizarComSheets() {
            if(!SHEETS_URL || SHEETS_URL.includes('SEU_SCRIPT_ID_AQUI')) {
                updateSyncBadge('noconfig');
                return;
            }
            if(!navigator.onLine) {
                updateSyncBadge('offline');
                return;
            }

            const pendentes = recordsData.filter(r => !r.sync);
            if(pendentes.length === 0) {
                updateSyncBadge('ok');
                return;
            }

            try {
                updateSyncBadge('enviando', pendentes.length);
                const res = await fetch(SHEETS_URL, {
                    method: 'POST',
                    headers: {'Content-Type': 'text/plain;charset=utf-8'},
                    body: JSON.stringify(pendentes)
                });
                const txt = await res.text();
                if(res.ok && txt.trim() === 'ok') {
                    pendentes.forEach(p => p.sync = true);
                    localStorage.setItem('agro_records', JSON.stringify(recordsData));
                    updateSyncBadge('ok');
                } else {
                    throw new Error(txt);
                }
            } catch(e) {
                updateSyncBadge('erro', pendentes.length);
            }
        }

        function updateSyncBadge(estado, qtd=0) {
            const badge = document.getElementById('sync-status-badge');
            if(!badge) return;
            if(estado === 'ok') {
                badge.className = 'text-xs bg-emerald-700 text-white font-bold px-3 py-1.5 rounded-full flex items-center gap-1.5 shadow-inner';
                badge.innerHTML = '<i class="fa-solid fa-cloud-arrow-up text-emerald-300"></i> Sincronizado ✔';
            } else if(estado === 'enviando') {
                badge.className = 'text-xs bg-amber-600 text-white font-bold px-3 py-1.5 rounded-full flex items-center gap-1.5 shadow-inner animate-pulse';
                badge.innerHTML = `<i class="fa-solid fa-arrows-rotate animate-spin"></i> A enviar ${qtd}...`;
            } else if(estado === 'offline') {
                badge.className = 'text-xs bg-slate-600 text-slate-200 font-bold px-3 py-1.5 rounded-full flex items-center gap-1.5 shadow-inner';
                badge.innerHTML = '<i class="fa-solid fa-wifi-slash"></i> Offline (Guardado local)';
            } else if(estado === 'noconfig') {
                badge.className = 'text-xs bg-slate-700 text-slate-200 font-bold px-3 py-1.5 rounded-full flex items-center gap-1.5 shadow-inner';
                badge.innerHTML = '<i class="fa-solid fa-triangle-exclamation"></i> Sheets por configurar';
            } else {
                badge.className = 'text-xs bg-red-600 text-white font-bold px-3 py-1.5 rounded-full flex items-center gap-1.5 shadow-inner';
                badge.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i> ${qtd} por sincronizar`;
            }
        }

        window.addEventListener('online', sincronizarComSheets);

        // --- UTILITÁRIOS ---
        function formatDatePt(iso) {
            if(!iso) return '';
            const p = iso.split('-');
            if(p.length !== 3) return iso;
            return `${p[2]}/${p[1]}/${p[0]}`;
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = msg;
            toast.classList.remove('translate-y-28', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-28', 'opacity-0');
            }, 3000);
        }
